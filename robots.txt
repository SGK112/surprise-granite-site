# ⚠️ READ BEFORE EDITING (fixed 2026-07-28):
# A named "User-agent: X" group REPLACES the "*" group for that crawler — it does
# NOT add to it. Per RFC 9309, a crawler picks the single most-specific matching
# group and ignores every other group.
# This file previously had 29 per-bot groups (Googlebot, Bingbot, GPTBot, …) that
# each contained only "Allow: /". The effect was that Googlebot and Bingbot read
# ONLY "Allow: /" and silently ignored every Disallow below — /api/, /staff/,
# /account/, /cart/, /checkout/ and /product-review/ were all unenforced for the
# two crawlers that matter most.
# Those groups were removed. "*" already allows every crawler, so AI bots remain
# just as welcome. Do NOT re-add per-bot "Allow: /" groups.

# No bare "Allow: /" here on purpose: crawling is allowed by default, so it adds
# nothing — but Google resolves conflicts by LONGEST match while several naive
# crawlers and SEO tools use FIRST match, and under first-match a leading
# "Allow: /" swallows every Disallow below it. Omitting it is correct under both.
User-agent: *
Disallow: /log-in/
Disallow: /account/
Disallow: /cart/
Disallow: /checkout/
Disallow: /estimate/
Disallow: /staff/
# API endpoints are not pages — stop crawlers wasting budget 404ing on them
Disallow: /api/
# Review form: one noindexed page reached via ~4,900 unique query strings.
# Blocking it keeps crawl budget on the real product pages.
Disallow: /product-review/
# Finsweet/Webflow CMS pagination params. These were ~30% of the "Crawled - currently
# not indexed" bucket (GSC 2026-08-07): every page carried a hidden paginator emitting
# ?xxxxxxxx_page=2, and Google then COMBINED the params across widgets
# (?017144ac_page=4&5dd91d68_page=4), so the URL space exploded. The widgets are gone
# (scripts/strip-webflow-pagination.js) — this covers the thousands of legacy URLs
# Google already knows, which still return 200. None of them are indexed, so blocking
# costs no canonical signal.
Disallow: /*_page=

# Sitemap location
Sitemap: https://www.surprisegranite.com/sitemap.xml
